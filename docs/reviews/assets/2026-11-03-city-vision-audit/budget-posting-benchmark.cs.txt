  var step=typeof(FoodCommerce).GetMethod("StepBudget",BindingFlags.NonPublic|BindingFlags.Instance)!;

   s.Places.Add(new Place {Id="home",Kind=PlaceKind.Home,Capacity=count});
   for(int i=0;i<count;i++) {
    var c=new CitizenState {Identity=new CitizenIdentity {Id="r"+i,Persona=Persona.Student,Age=22},HomeId="home",PlaceId="home",Activity=Activity.Home,Phase=CitizenPhase.AtPlace,NextEventMinute=2879,ActivityUntilMinute=2879};
