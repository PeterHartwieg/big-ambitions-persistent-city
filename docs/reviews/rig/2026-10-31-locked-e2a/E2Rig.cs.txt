// rig_test stage only. All reads and bounded watch samples run on Unity's main thread.
using System;
using System.IO;
using System.Linq;
using System.Collections;
using System.Collections.Generic;
using System.Reflection;
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using PersistentCity.Core;
using PersistentCity.Runtime;
using UnityEngine;
using BigAmbitions.DayNightCycle;
namespace PerformanceReview {
public class E2Runner:MonoBehaviour {
 public string Root=""; public static E2Runner Owner; bool watching; float next; int sample; string prefix;
 static BindingFlags F=BindingFlags.Instance|BindingFlags.Static|BindingFlags.Public|BindingFlags.NonPublic;
 void OnEnable(){Owner=this;}
 static object Get(object v,string n){if(v==null)return null;var t=v.GetType();var f=t.GetField(n,F);if(f!=null)return f.GetValue(v);var p=t.GetProperty(n,F);return p==null||p.GetIndexParameters().Length!=0?null:p.GetValue(v);}
 static object Scalar(object v){if(v==null)return null;if(v is UnityEngine.Object o)return new{type=v.GetType().FullName,name=o.name,instance=o.GetInstanceID()};return v;}
 static Dictionary<string,object> Read(object v,params string[] names){return names.ToDictionary(n=>n,n=>Scalar(Get(v,n)));}
 static PopulationRuntime Runtime()=>UnityEngine.Object.FindObjectOfType<PopulationRuntime>();
 static CitySimulator Sim()=>(CitySimulator)Get(Runtime(),"simulation");
 static object Stop(OptionalStopState s)=>s==null?null:new{kind=s.Kind.ToString(),s.PlaceId,s.SpotId,s.TargetId,s.Position,s.Facing,s.ReturnPosition,s.StartMinute,s.EndMinute,s.VisitMinute,s.ResumeEventMinute,s.Cancelled,s.Members};
 public static object Snapshot(){var r=Runtime();var sim=Sim();var state=sim.State;var physical=Get(r,"physical");var bindings=Get(physical,"bindings") as IDictionary;var bodies=new List<object>();
 if(bindings!=null)foreach(DictionaryEntry item in bindings){var b=item.Value;bodies.Add(new{id=item.Key,step=Get(b,"Step")?.ToString(),optionalReturning=Get(b,"OptionalReturning"),slot=Get(b,"Slot"),queue=Get(b,"Queue"),optional=Stop((OptionalStopState)Get(b,"Optional")),person=Scalar(Get(Get(b,"Person"),"Character")),car=Get(b,"Car")!=null});}
 var stops=state.Citizens.Where(c=>c.OptionalStop!=null).Select(c=>new{id=c.Identity.Id,assigned=c.DayPlaceId,current=c.PlaceId,activity=c.Activity.ToString(),phase=c.Phase.ToString(),c.ActivityUntilMinute,c.NextEventMinute,c.OptionalStopChoice,stop=Stop(c.OptionalStop),status=c.OptionalStop.Cancelled?"cancelled":state.Minute<c.OptionalStop.StartMinute?"future":c.OptionalStop.ActiveAt(state.Minute)?"active":"expired",meal=c.MealOuting!=null}).ToArray();
 var queues=new List<object>();foreach(var c in state.Citizens){var p=state.Places.FirstOrDefault(x=>x.Id==c.PlaceId);if(p==null)continue;var d=DwellPlanner.QueueAtOpening(c,p,state.Seed,state.Minute,RainHelper.isRaining);if(d!=null)queues.Add(new{id=c.Identity.Id,assigned=c.DayPlaceId,current=c.PlaceId,activity=c.Activity.ToString(),phase=c.Phase.ToString(),c.ActivityUntilMinute,c.NextEventMinute,dwell=d});}
 var spots=Get(physical,"spots");var cached=Get(spots,"queues") as IDictionary;var lines=new List<object>();if(cached!=null)foreach(DictionaryEntry v in cached)lines.Add(new{key=v.Key,line=v.Value});
 return new{frame=Time.frameCount,state.Minute,state.Seed,state.Schema,population=state.Citizens.Count,identityDigest=Digest(string.Join("\n",state.Citizens.Select(c=>c.Identity.Id).OrderBy(x=>x))),pending=state.Citizens.Count(c=>c.OptionalStopChoice==true&&c.OptionalStop==null),tally=sim.OptionalStopChances,raining=RainHelper.isRaining,stops,queues,bodies,physical=Get(physical,"Diagnostics"),pedestrianLimit=Get(physical,"pedestrianLimit"),vehicleLimit=Get(physical,"vehicleLimit"),cachedLines=lines,foodShops=state.Food?.Shops.Select(s=>s.PlaceId).ToArray(),meals=state.Citizens.Where(c=>c.MealOuting!=null).Select(c=>new{id=c.Identity.Id,c.PlaceId,c.ActivityUntilMinute,outing=c.MealOuting}).ToArray(),truncated=0};}
 static string Digest(string s){using(var h=System.Security.Cryptography.SHA256.Create())return BitConverter.ToString(h.ComputeHash(System.Text.Encoding.UTF8.GetBytes(s))).Replace("-","").ToLowerInvariant();}
 static object Venues(){var sim=Sim();string[] types={"coffeeshop","fastfoodrestaurant","nightclub","gym","cinema","theater","lawfirm","school","warehouse"};var asm=typeof(SaveGameManager).Assembly;var type=asm.GetTypes().FirstOrDefault(t=>t.Name=="BusinessType");var assets=type==null?new UnityEngine.Object[0]:Resources.FindObjectsOfTypeAll(type);var catalog=assets.Select(a=>new{asset=Scalar(a),facts=Read(a,"id","name","businessTypeName","hasPedestriansOutside","pedestrianPool")}).ToArray();var controllers=CityManager.Instance.cityBuildingControllers.Where(c=>c!=null).ToArray();var rows=new List<object>();
 foreach(var c in controllers){var reg=c.buildingRegistration;if(reg==null||!types.Any(t=>reg.businessTypeName=="ba:businesstype_"+t))continue;var doors=c.entranceDoors?.Where(d=>d!=null&&d.doorTransform!=null).Select(d=>new{instance=d.doorTransform.GetInstanceID(),position=d.doorTransform.position.ToString(),forward=d.doorTransform.forward.ToString()}).ToArray();var address=reg.StreetName+":"+reg.StreetNumber;rows.Add(new{type=reg.businessTypeName,controller=c.GetInstanceID(),active=c.gameObject.activeInHierarchy,address,reg.HasEstablishedBusiness,reg.AvailableForRent,hours=reg.scheduleDays,closure=Read(reg,"Closed","IsClosed","closed","isClosed","isTemporarilyClosed","temporarilyClosed"),addressControllerCount=controllers.Count(x=>x.buildingRegistration!=null&&x.buildingRegistration.StreetName==reg.StreetName&&x.buildingRegistration.StreetNumber==reg.StreetNumber),doors,places=sim.State.Places.Where(p=>p.Id.StartsWith("address:"+Uri.EscapeDataString(reg.StreetName)+":"+reg.StreetNumber+":",StringComparison.Ordinal)).Select(p=>new{p.Id,kind=p.Kind.ToString(),p.Position,p.WeeklyHours}).ToArray()});}
 var nativeTypes=asm.GetTypes().Where(t=>new[]{"BuildingOutsideHangoutZoneController","BuildingOutsideHangoutZone","AiSpawnerForBuildingOutsideHangoutZone","BuildingOutsidePedestrian","StationaryAiSpawner"}.Contains(t.Name)).ToArray();var actors=new List<object>();foreach(var t in nativeTypes){if(!typeof(UnityEngine.Object).IsAssignableFrom(t))continue;foreach(var v in Resources.FindObjectsOfTypeAll(t))actors.Add(new{type=t.FullName,instance=v.GetInstanceID(),name=v.name,active=v is Component cc?(bool?)cc.gameObject.activeInHierarchy:null,position=v is Component co?co.transform.position.ToString():null,ownership="native object; visual overlap unverified"});}
 return new{frame=Time.frameCount,requestedTypes=types,catalog,venues=rows,nativeActors=actors,weatherMembers=typeof(RainHelper).GetMembers(F).Where(m=>m.MemberType==MemberTypes.Field||m.MemberType==MemberTypes.Method).Select(m=>m.ToString()).ToArray(),readOnly=true};}
 public static object Dispatch(JObject q){string action=(string)q["action"];if(action=="provenance"){var asm=typeof(PopulationRuntime).Assembly;string sha;using(var h=System.Security.Cryptography.SHA256.Create())sha=BitConverter.ToString(h.ComputeHash(File.ReadAllBytes(asm.Location))).Replace("-","").ToLowerInvariant();return new{asm.FullName,location=asm.Location,sha256=sha};}if(action=="snapshot")return Snapshot();if(action=="venues")return Venues();if(action=="watch"){Owner.watching=(bool?)q["enabled"]??true;Owner.prefix=(string)q["prefix"]??"watch";Owner.sample=0;return new{Owner.watching};}throw new Exception("Unknown E2a action");}
 void Update(){if(!watching||Time.unscaledTime<next)return;next=Time.unscaledTime+2;try{File.WriteAllText(Path.Combine(Root,"e2-"+prefix+"-"+(sample++).ToString("D4")+".json"),JsonConvert.SerializeObject(Snapshot()));}catch(Exception e){watching=false;File.WriteAllText(Path.Combine(Root,"e2-watch-error.json"),JsonConvert.SerializeObject(new{error=e.ToString()}));}}
}}
