static class SelectionPerf {
 static void Measure(string name,Action action) {
  for(int i=0;i<100;i++)action();
  var times=new double[300]; long bytes=GC.GetAllocatedBytesForCurrentThread(); int spikes=0;
  for(int i=0;i<times.Length;i++) {
   long start=System.Diagnostics.Stopwatch.GetTimestamp();action();
   times[i]=(System.Diagnostics.Stopwatch.GetTimestamp()-start)*1000.0/System.Diagnostics.Stopwatch.Frequency;
   if(times[i]>=30)spikes++;
  }
  bytes=GC.GetAllocatedBytesForCurrentThread()-bytes;Array.Sort(times);
  Console.WriteLine($"offline {name}: ms/update p50={times[150]:F6} p95={times[285]:F6} max={times[299]:F6}; bytes/update={bytes/300.0:F1}; spikes>=30ms={spikes}; 300 single calls (native doubles).");
 }
 public static void Run() {
  var before=new BeforeF0();var after=new After();
  foreach(Fixture f in new Fixture[]{before,after}) {
   f.InitialPeople=f.pedestrianLimit=48;f.InitialCars=f.vehicleLimit=24;
   for(int i=0;i<957;i++) {
    var leg=new RouteLeg {Mode=TravelMode.Walk,From=new Position(i%50,0,0),To=new Position(i%50+5,0,0),Minutes=1};
    var c=new CitizenState {Identity=new CitizenIdentity{Id=i.ToString()},Phase=CitizenPhase.Travelling,Journey=new Route{Legs=new(){leg}},NextEventMinute=200};
    f.simulation.State.Citizens.Add(c);f.simulation.Samples[c.Identity.Id]=new PhysicalJourneySample(0,1,0,0,leg,leg.From,null,null);
   }
  }
  Measure("957-resident full-pool selection BEFORE F0 cheap break",()=>before.Select());
  Measure("957-resident full-pool selection AFTER F0 cheap break",()=>after.Select());
  F5f();
  F2f();
 }
 // F2f: open pools (48/24), N persons of which A are active, spread over a 3 km x 3 km city on two-leg walks of
 // about 150-450 m (a third of them waiting at a door or at an outdoor scene instead); the focus is at the
 // centre with the 120 m street radius. F5F is the merged F5f index selection (BeforeF2f), GRID this build.
 // DENSE puts every active person inside the radius, the grid's worst case.
 static void F2f() {
  foreach (bool dense in new[]{false,true}) foreach (int n in new[]{1914,10000,50000}) foreach (int active in new[]{100,1000,5000}) {
   if (dense && n != 10000) continue;
   var rng=new Random(n+active);
   var f5f=new BeforeF2f();var grid=new After();
   double span=dense?150:3000;
   foreach(Fixture f in new Fixture[]{f5f,grid}) {
    f.pedestrianLimit=48;f.vehicleLimit=24;f.radius=120;f.focus=new Position(0,0,0);
    f.places["indoor"]=new Place{Id="indoor",Position=new Position(span*2,0,span*2)};
    for(int p=0;p<60;p++) f.places["scene-door"+p]=new Place{Id="scene-door"+p,Position=new Position((p*37%60/60.0-.5)*span,0,(p*53%60/60.0-.5)*span)};
   }
   var people=new List<(string Id,Route? Route,string Place,CitizenPhase Phase)>();
   for(int i=0;i<n;i++) {
    string id=i.ToString("D5");
    if(i>=active) { people.Add((id,null,"indoor",CitizenPhase.AtPlace)); continue; }
    if(i%3==2) { people.Add((id,null,"scene-door"+(i%60),i%2==0?CitizenPhase.WaitingForOpening:CitizenPhase.AtPlace)); continue; }
    var a=new Position((rng.NextDouble()-.5)*span,0,(rng.NextDouble()-.5)*span);
    var b=new Position(a.X+(rng.NextDouble()-.5)*300,0,a.Z+(rng.NextDouble()-.5)*300);
    var c=new Position(b.X+(rng.NextDouble()-.5)*300,0,b.Z);
    people.Add((id,new Route{Legs=new(){new RouteLeg{Mode=TravelMode.Walk,From=a,To=b,Minutes=3},new RouteLeg{Mode=TravelMode.Walk,From=b,To=c,Minutes=3}}},"indoor",CitizenPhase.Travelling));
   }
   foreach(Fixture f in new Fixture[]{f5f,grid}) {
    foreach(var p in people) {
     // Door waits and outdoor scene visitors share scene places; queue entries are pre-planned for both arms.
     var c=new CitizenState{Identity=new CitizenIdentity{Id=p.Id},Phase=p.Phase,Journey=p.Route,PlaceId=p.Place,
      Activity=p.Phase==CitizenPhase.AtPlace&&p.Place!="indoor"?Activity.Leisure:Activity.Work,NextEventMinute=200};
     if(p.Phase==CitizenPhase.WaitingForOpening) f.queuers[p.Id]=new QueuePlanEntry();
     f.simulation.State.Citizens.Add(c);
     if(p.Route!=null) { var leg=p.Route.Legs[0]; f.simulation.Samples[p.Id]=new PhysicalJourneySample(0,1,0,0,leg,leg.From,null,null); }
    }
    _=f.simulation.ActiveJourneys;
   }
   f5f.attempted.Clear();grid.attempted.Clear();
   var expected=f5f.Select();
   Check(expected.SequenceEqual(grid.Select()),"F2f perf fixture selections differ");
   var index=grid.simulation.ActiveJourneys;var nearby=new List<CitizenState>();
   int returned=index.Grid.Query(grid.focus,grid.radius,nearby).Count;
   string label=$"F2f {(dense?"DENSE ":"")}{n} persons / {active} active ({index.BodyCandidates.Count} indexed, {returned} from grid cells, {index.Grid.Registrations} cell registrations)";
   Measure(label+" selection F5F (index)",()=>{f5f.attempted.Clear();f5f.Select();});
   Measure(label+" selection GRID",()=>{grid.attempted.Clear();grid.Select();});
   Measure(label+" grid query only",()=>index.Grid.Query(grid.focus,grid.radius,nearby));
   var roster=grid.simulation.State.Citizens;
   Measure(label+" shadow comparison with grid (30 s cadence)",()=>{if(!index.MatchesScan(roster))throw new Exception("shadow");});
   if(!dense && active==1000) {
    // One route commit: a travelling member's journey is replaced by a fresh two-leg route and re-filed.
    var probe=roster.First(c=>c.Phase==CitizenPhase.Travelling);
    var routes=Enumerable.Range(0,2).Select(k=>new Route{Legs=new(){
     new RouteLeg{Mode=TravelMode.Walk,From=new Position(k*10,0,0),To=new Position(k*10+250,0,120),Minutes=3},
     new RouteLeg{Mode=TravelMode.Walk,From=new Position(k*10+250,0,120),To=new Position(k*10+250,0,400),Minutes=3}}}).ToArray();
    int flip=0;
    Measure($"F2f {n} persons grid filing per route commit (two-leg ~530 m route)",()=>{probe.Journey=routes[++flip%2];index.File(probe);});
    Measure($"F2f {n} persons index rebuild with grid (construction / load)",()=>index.Rebuild(roster));
   }
  }
 }
 // F5f: open pools (48/24), N persons of which A walk within the radius; the rest are indoors at work.
 // BEFORE is the unchanged pre-F5f loop over every citizen (BeforeF5f: no index, no ID classification);
 // AFTER iterates the active-journey index; FALLBACK is this build's full scan after a shadow mismatch,
 // which also classifies every ID to withhold visitors. Each call clears cooldowns so it attempts again.
 static void F5f() {
  foreach (int n in new[]{1914,10000,50000}) foreach (int active in new[]{0,100,1000}) {
   var baseline=new BeforeF5f();var full=new After{journeyIndexFallback=true};var indexed=new After();
   foreach(Fixture f in new Fixture[]{baseline,full,indexed}) {
    f.pedestrianLimit=48;f.vehicleLimit=24;f.places["indoor"]=new Place{Id="indoor"};
    for(int i=0;i<n;i++) {
     var id=i.ToString();
     if(i<active) {
      var leg=new RouteLeg {Mode=TravelMode.Walk,From=new Position(i%90,0,0),To=new Position(i%90+5,0,0),Minutes=1};
      f.simulation.State.Citizens.Add(new CitizenState {Identity=new CitizenIdentity{Id=id},Phase=CitizenPhase.Travelling,Journey=new Route{Legs=new(){leg}},NextEventMinute=200});
      f.simulation.Samples[id]=new PhysicalJourneySample(0,1,0,0,leg,leg.From,null,null);
     } else f.simulation.State.Citizens.Add(new CitizenState {Identity=new CitizenIdentity{Id=id},Phase=CitizenPhase.AtPlace,Activity=Activity.Work,PlaceId="indoor",NextEventMinute=200});
    }
    _=f.simulation.ActiveJourneys;
   }
   baseline.attempted.Clear();full.attempted.Clear();indexed.attempted.Clear();
   var expected=baseline.Select();
   Check(expected.SequenceEqual(indexed.Select()) && expected.SequenceEqual(full.Select()),"F5f perf fixture selections differ");
   Measure($"F5f {n} persons / {active} active journeys selection BEFORE (pre-F5f loop)",()=>{baseline.attempted.Clear();baseline.Select();});
   Measure($"F5f {n} persons / {active} active journeys selection AFTER (index)",()=>{indexed.attempted.Clear();indexed.Select();});
   Measure($"F5f {n} persons / {active} active journeys selection FALLBACK (full scan after mismatch)",()=>{full.attempted.Clear();full.Select();});
   var index=indexed.simulation.ActiveJourneys;var roster=indexed.simulation.State.Citizens;
   Measure($"F5f {n} persons / {active} active journeys shadow comparison (30 s cadence)",()=>{if(!index.MatchesScan(roster))throw new Exception("shadow");});
   if(active==1000) {
    var probe=roster[n-1];
    Measure($"F5f {n} persons index membership change (one phase change in and out)",()=>{
     probe.Phase=CitizenPhase.Travelling;index.File(probe);probe.Phase=CitizenPhase.AtPlace;index.File(probe);});
    Measure($"F5f {n} persons index rebuild (construction / load)",()=>index.Rebuild(roster));
   }
  }
 }
 static void Check(bool condition,string text){if(!condition)throw new Exception(text);}
}
