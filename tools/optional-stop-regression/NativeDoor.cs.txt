using System;
using System.Collections.Generic;
using PersistentCity.Core;
namespace PersistentCity.Runtime
{
    sealed class CityBuildingController
    {
        public bool Destroyed;
        public Position Entrance;
        public GameObject gameObject = new GameObject();
        public Registration buildingRegistration = new Registration();
        public static bool operator ==(CityBuildingController? a, CityBuildingController? b)
            => ReferenceEquals(a,b) || (ReferenceEquals(a,null) || a.Destroyed) && (ReferenceEquals(b,null) || b.Destroyed);
        public static bool operator !=(CityBuildingController? a, CityBuildingController? b) => !(a==b);
        public override bool Equals(object? o) => ReferenceEquals(this,o);
        public override int GetHashCode() => base.GetHashCode();
    }
    sealed class GameObject { public bool activeInHierarchy = true; }
    sealed class Registration
    {
        public bool HasEstablishedBusiness = true;
        public bool temporarilyClosed = false;
        public string businessTypeName = "ba:businesstype_nightclub";
    }
    sealed class CityManager
    {
        public static CityManager Instance = new CityManager();
        public List<CityBuildingController> cityBuildingControllers = new List<CityBuildingController>();
    }
    static class GamePlaces
    {
        public static bool TryEntrance(CityBuildingController c, out Position p) { p=c.Entrance; return true; }
        public static bool BuildingAccessible(CityBuildingController c) => true;
    }
    sealed class NativeDoorHarness
    {
        readonly DwellSpots doors = new DwellSpots();
        readonly Dictionary<string,CityBuildingController> buildings = new Dictionary<string,CityBuildingController>();
        readonly Dictionary<string,OptionalStopSpot[]> cached = new Dictionary<string,OptionalStopSpot[]>();
        readonly Dictionary<string,OptionalStopSpot> validatedSpots = new Dictionary<string,OptionalStopSpot>();
        /* PRODUCTION METHOD */
        public static void Run()
        {
            var h=new NativeDoorHarness(); var p=new Place {Id="shop", Position=new Position(1,0,1)};
            var old=new CityBuildingController {Entrance=p.Position}; CityManager.Instance.cityBuildingControllers.Add(old);
            if (!h.NativeDoorPresent(p,DwellKind.WindowShopping)) throw new Exception("Initial native door missing");
            Time.frameCount++; h.doors.TryQueue(p,out var line);
            var spot=new OptionalStopSpot {Id="old-spot"};
            h.cached[p.Id+":"+DwellKind.WindowShopping]=new[] {spot}; h.validatedSpots[spot.Id]=spot;
            old.Destroyed=true; var replacement=new CityBuildingController {Entrance=p.Position};
            CityManager.Instance.cityBuildingControllers.Clear(); CityManager.Instance.cityBuildingControllers.Add(replacement);
            if (!h.NativeDoorPresent(p,DwellKind.WindowShopping) || !ReferenceEquals(h.buildings[p.Id],replacement))
                throw new Exception("Destroyed cached native door must be replaced");
            if (h.cached.Count!=0 || h.validatedSpots.Count!=0 || !h.doors.Unsampled(p))
                throw new Exception("Native replacement must invalidate all derived facade geometry");
            Time.frameCount++; h.doors.TryQueue(p,out var newLine);
            if (ReferenceEquals(line,newLine)) throw new Exception("Replacement reused obsolete facade");
            replacement.Destroyed=true; CityManager.Instance.cityBuildingControllers.Clear();
            if (h.NativeDoorPresent(p,DwellKind.WindowShopping)) throw new Exception("Missing replacement must fail closed");
            CityManager.Instance.cityBuildingControllers.Add(new CityBuildingController {Entrance=p.Position});
            if (!h.NativeDoorPresent(p,DwellKind.WindowShopping)) throw new Exception("Later replacement must remain discoverable");
            Console.WriteLine("Native door replacement: 6 checks passed.");
        }
    }
}
