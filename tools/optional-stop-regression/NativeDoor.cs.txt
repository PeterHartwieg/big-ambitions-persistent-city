        /* PRODUCTION METHOD */
        static void ReloadMultipleEntrances(bool reverse)
        {
            CityManager.Instance.cityBuildingControllers.Clear(); Time.frameCount++;
            var home=new Place {Id="home",Name="Home",Kind=PlaceKind.Home,Capacity=2};
            var work1=new Place {Id="work1",Name="Work 1",Kind=PlaceKind.Work,Capacity=2,Position=new Position(10,0,0)};
            var work2=new Place {Id="work2",Name="Work 2",Kind=PlaceKind.Work,Capacity=2,Position=new Position(20,0,0)};
            var state=Population.Create(new[] {new CitizenIdentity {Id="a",Name="A",Age=30},new CitizenIdentity {Id="b",Name="B",Age=30}},
                new[] {home,work1,work2},"saved-stops","0.40.4",17,100);
            for(int i=0;i<2;i++)
            {
                var c=state.Citizens[i]; var p=i==0?work1:work2;
                c.Phase=CitizenPhase.AtPlace; c.PlaceId=p.Id; c.Activity=Activity.Work;
                c.NextEventMinute=c.ActivityUntilMinute=1000; c.OptionalStopChoice=true;
                c.RecentVisits.Add(new VisitRecord {AtMinute=100,PlaceId=p.Id,Activity=Activity.Work});
                c.OptionalStop=new OptionalStopState {Kind=DwellKind.Break,PlaceId=p.Id,TargetId=p.Id,SpotId=p.Id+":Break:0",
                    Position=p.Position,Facing=p.Position,ReturnPosition=p.Position,VisitMinute=100,StartMinute=160,EndMinute=165,ResumeEventMinute=1000};
                CityManager.Instance.cityBuildingControllers.Add(new CityBuildingController {Entrance=p.Position});
            }
            if(reverse) state.Citizens.Reverse();
            string saved=SaveCodec.Write(state);
            var read=SaveCodec.Read(saved); if(read.Status!=SaveReadStatus.Loaded) throw new Exception(read.Error);
            var env=new NativeDoorHarness(); foreach(var p in read.State!.Places) env.places[p.Id]=p;
            var sim=new CitySimulator(read.State,new NoRoutes()); int before=DwellSpots.SampleCalls;
            sim.AttachOptionalStops(env,25);
            if(read.State.Citizens.Any(c=>c.OptionalStop!.Cancelled) || SaveCodec.Write(read.State)!=saved)
                throw new Exception("Attach cancelled or modified a saved future break");
            var second=read.State.Citizens[1].OptionalStop!;
            if(sim.OptionalStopStatus(second)!=OptionalStopAvailability.Deferred || DwellSpots.SampleCalls!=before+1)
                throw new Exception("Cold second entrance must defer without exceeding frame budget");
            // Active saved records likewise cannot gain a body before validation.
            read.State.Minute=160;
            if(Harness.TryDeferredAdmission(sim,read.State.Citizens[1])) throw new Exception("Unvalidated stop received a body");
            Time.frameCount++; sim.RefreshOptionalStops();
            if(sim.OptionalStopStatus(second)!=OptionalStopAvailability.Ready || read.State.Citizens.Any(c=>c.OptionalStop!.Cancelled)
                || DwellSpots.SampleCalls!=before+2) throw new Exception("Next frame failed to resume validation");
            if(!Harness.TryDeferredAdmission(sim,read.State.Citizens[1])) throw new Exception("Validated stop cannot be joined");
            if(second.StartMinute!=160 || second.EndMinute!=165 || read.State.Citizens.Any(c=>c.OptionalStopChoice!=true))
                throw new Exception("Validation changed saved choice or deadline");
            // A fresh reload while rain starts must cancel even the entrance still deferred.
            var wet=SaveCodec.Read(saved).State!; var wetEnv=new NativeDoorHarness();
            foreach(var p in wet.Places) wetEnv.places[p.Id]=p;
            var wetSim=new CitySimulator(wet,new NoRoutes()); wetSim.AttachOptionalStops(wetEnv,25);
            wetEnv.Raining=true; wetSim.RefreshOptionalStops();
            if(wet.Citizens.Any(c=>!c.OptionalStop!.Cancelled)) throw new Exception("Rain must cancel deferred future stops");
            // Confirmed native loss is different from budget deferral.
            var native=CityManager.Instance.cityBuildingControllers.Single(b=>b.Entrance.Equals(second.ReturnPosition));
            native.Destroyed=true; CityManager.Instance.cityBuildingControllers.Remove(native);
            sim.RefreshOptionalStops();
            if(!second.Cancelled) throw new Exception("Confirmed disappearance must cancel");
            env.Raining=true; sim.RefreshOptionalStops();
            if(read.State.Citizens.Any(c=>!c.OptionalStop!.Cancelled)) throw new Exception("Rain must still cancel");
            Console.WriteLine("Saved facade reload: 9 integration checks passed.");
        }

        public static void Run()
        {
            VenueBoundaries();
            Findings();
            IndexedBoundaries();
            ReloadMultipleEntrances(false);
            ReloadMultipleEntrances(true);
            var h=new NativeDoorHarness(); var p=new Place {Kind=PlaceKind.Work,Id="shop", Position=new Position(1,0,1)};
            var old=new CityBuildingController {Entrance=p.Position}; CityManager.Instance.cityBuildingControllers.Add(old);
            if (!h.NativeDoorPresent(p,DwellKind.WindowShopping)) throw new Exception("Initial native door missing");
            Time.frameCount++; h.doors.TryQueue(p,out var line);
            var spot=new OptionalStopSpot {Id="old-spot"};
            h.cached[p.Id+":"+DwellKind.WindowShopping]=new[] {spot}; h.validatedSpots[spot.Id]=spot;
            old.Destroyed=true; var replacement=new CityBuildingController {Entrance=p.Position};
            CityManager.Instance.cityBuildingControllers.Clear(); CityManager.Instance.cityBuildingControllers.Add(replacement);
            if (!h.NativeDoorPresent(p,DwellKind.WindowShopping) || !ReferenceEquals(h.buildings[p.Id],replacement))
                throw new Exception("Destroyed cached native door must be replaced");
            if (h.cached.Count!=0 || h.validatedSpots.Count!=0 || !h.doors.Unsampled(p))
                throw new Exception("Native replacement must invalidate all derived facade geometry");
            Time.frameCount++; h.doors.TryQueue(p,out var newLine);
            if (ReferenceEquals(line,newLine)) throw new Exception("Replacement reused obsolete facade");
            replacement.Destroyed=true; CityManager.Instance.cityBuildingControllers.Clear();
            if (h.NativeDoorPresent(p,DwellKind.WindowShopping)) throw new Exception("Missing replacement must fail closed");
            CityManager.Instance.cityBuildingControllers.Add(new CityBuildingController {Entrance=p.Position});
            if (h.NativeDoorPresent(p,DwellKind.WindowShopping)) throw new Exception("Cached refusal must not poll for later replacements");
            h=new NativeDoorHarness();
            if (!h.NativeDoorPresent(p,DwellKind.WindowShopping)) throw new Exception("Republished environment must discover replacement");
            Console.WriteLine("Native door replacement/refusal republication: 7 checks passed.");
        }

        static void VenueBoundaries()
        {
            int checks=0;
            // Native identifier evidence is documented separately. These doubles test only
            // the unchanged predicate, not asset availability or new chat eligibility.
            foreach(string type in new[] {"nightclub","coffeeshop","fastfoodrestaurant","gym","cinema","theater","school","lawfirm","warehouse","unknown"})
            {
                var p=new Place {Kind=PlaceKind.Work,Id="venue:"+type,Position=new Position(40,0,0)};
                var b=new CityBuildingController {Entrance=p.Position};
                b.buildingRegistration!.businessTypeName="ba:businesstype_"+type;
                CityManager.Instance.cityBuildingControllers.Clear(); CityManager.Instance.cityBuildingControllers.Add(b);
                var h=new NativeDoorHarness();
                if(!h.NativeDoorPresent(p,DwellKind.Break) || !h.NativeDoorPresent(p,DwellKind.WindowShopping)
                    || h.NativeDoorPresent(p,DwellKind.ChatOutsideVenue)!=(type=="nightclub")) throw new Exception("Venue predicate changed: "+type);
                checks+=3;
                foreach(Action<CityBuildingController> damage in new Action<CityBuildingController>[] {
                    x=>x.gameObject.activeInHierarchy=false, x=>x.Accessible=false, x=>x.HasEntrance=false,
                    x=>x.buildingRegistration=null, x=>x.buildingRegistration!.HasEstablishedBusiness=false,
                    x=>x.buildingRegistration!.temporarilyClosed=true, x=>x.buildingRegistration!.StreetName="" })
                {
                    var bad=new CityBuildingController {Entrance=p.Position};
                    bad.buildingRegistration!.businessTypeName="ba:businesstype_"+type;
                    damage(bad);
                    CityManager.Instance.cityBuildingControllers.Clear(); CityManager.Instance.cityBuildingControllers.Add(bad);
                    var missing=new NativeDoorHarness();
                    foreach(var kind in new[] {DwellKind.Break,DwellKind.WindowShopping,DwellKind.ChatOutsideVenue})
                    { if(missing.NativeDoorPresent(p,kind)) throw new Exception("Missing venue fact accepted"); checks++; }
                }
            }
            CityManager.Instance.cityBuildingControllers.Clear();
            Console.WriteLine($"Native venue boundaries: {checks} checks passed (doubles; no venue enabled).");
        }

        public static void Findings()
        {
            var p=new Place {Kind=PlaceKind.Work,Id="ambiguous",Position=new Position(40,0,0)};
            var club=new CityBuildingController {Entrance=p.Position};
            var cafe=new CityBuildingController {Entrance=p.Position}; cafe.buildingRegistration!.businessTypeName="ba:businesstype_coffeeshop";
            CityManager.Instance.cityBuildingControllers.Clear(); CityManager.Instance.cityBuildingControllers.AddRange(new[] {club,cafe});
            var ambiguous=new NativeDoorHarness();
            bool clubFirst=ambiguous.NativeDoorPresent(p,DwellKind.ChatOutsideVenue);
            Time.frameCount++; int samples=DwellSpots.SampleCalls;
            if(ambiguous.Spots(p,DwellKind.Break).Count!=0 || DwellSpots.SampleCalls!=samples)
                throw new Exception("Refused frontage consumed facade sample budget");
            CityManager.Instance.cityBuildingControllers.Reverse();
            bool cafeFirst=new NativeDoorHarness().NativeDoorPresent(p,DwellKind.ChatOutsideVenue);
            if(clubFirst || cafeFirst) throw new Exception("R2 shared frontage must fail closed in both orders");
            var cachedRefusal=new NativeDoorHarness();
            if(cachedRefusal.NativeDoorPresent(p,DwellKind.Break)) throw new Exception("Ambiguous break accepted");
            CityManager.Instance.cityBuildingControllers.Remove(cafe);
            if(cachedRefusal.NativeDoorPresent(p,DwellKind.Break)) throw new Exception("Refusal was polled again");
            if(!new NativeDoorHarness().NativeDoorPresent(p,DwellKind.ChatOutsideVenue)) throw new Exception("Unique match changed");
            CityManager.Instance.cityBuildingControllers.Add(club);
            if(!new NativeDoorHarness().NativeDoorPresent(p,DwellKind.ChatOutsideVenue)) throw new Exception("Repeated reference is not a second owner");
            var addressed=new Place {Kind=PlaceKind.Work,Id="address:Test%20Street:1:work",Position=p.Position};
            if(!new NativeDoorHarness().NativeDoorPresent(addressed,DwellKind.Break)) throw new Exception("Unique address refused");
            club.buildingRegistration!.StreetNumber=2;
            if(new NativeDoorHarness().NativeDoorPresent(addressed,DwellKind.Break)) throw new Exception("Conflicting address accepted");
            club.buildingRegistration.StreetNumber=1;
            cafe.Entrance=new Position(80,0,0);
            CityManager.Instance.cityBuildingControllers.Add(cafe);
            foreach(bool reverse in new[]{false,true})
            {
                if(reverse) CityManager.Instance.cityBuildingControllers.Reverse();
                if(new NativeDoorHarness().NativeDoorPresent(addressed,DwellKind.Break)) throw new Exception("Duplicate address at different entrance accepted");
            }
            Console.WriteLine($"R2 acceptance: ambiguous club-first={clubFirst}, cafe-first={cafeFirst}; cached refusal, unique/repeated reference and conflicting address checks passed.");
            CityManager.Instance.cityBuildingControllers.Clear();
        }

        static void IndexedBoundaries()
        {
            int checks=0;
            foreach(double origin in new[]{-10.00001,-0.100001,-0.000001,0.099999,10.00001})
                foreach(int axis in new[]{0,1,2})
                    foreach(double offset in new[]{-0.10001,-0.09999,0.09999,0.10001})
                    {
                        var p=new Place {Id="address:Test%20Street:1:work",Kind=PlaceKind.Work,Position=new Position(origin,origin,origin)};
                        var own=new CityBuildingController {Entrance=p.Position};
                        var other=new CityBuildingController {Entrance=new Position(origin+(axis==0?offset:0),origin+(axis==1?offset:0),origin+(axis==2?offset:0))};
                        other.buildingRegistration!.StreetNumber=2;
                        foreach(bool reverse in new[]{false,true})
                        {
                            CityManager.Instance.cityBuildingControllers.Clear();
                            CityManager.Instance.cityBuildingControllers.AddRange(reverse?new[]{other,own}:new[]{own,other});
                            var h=new NativeDoorHarness(); h.Publish(new[]{p});
                            bool expected=Math.Abs(offset)>0.1;
                            if(h.NativeDoorPresent(p,DwellKind.Break)!=expected) throw new Exception("Entrance-cell edge changed ownership");
                            checks++;
                        }
                    }
            CityManager.Instance.cityBuildingControllers.Clear();
            var roster=Enumerable.Range(0,1000).Select(i=>new Place {Id="address:Test%20Street:"+i+":work",Kind=i<500?PlaceKind.Home:i<600?PlaceKind.Study:PlaceKind.Work,Position=new Position(i*10,0,0)}).ToArray();
            foreach(var p in roster)
            {
                var b=new CityBuildingController {Entrance=p.Position}; b.buildingRegistration!.StreetNumber=CityManager.Instance.cityBuildingControllers.Count;
                CityManager.Instance.cityBuildingControllers.Add(b);
            }
            var env=new NativeDoorHarness(); int before=GamePlaces.EntranceReads; env.Publish(roster);
            if(env.buildings.Count!=400 || GamePlaces.EntranceReads-before!=1000 || env.FrontageRebindCount!=0) throw new Exception("Non-frontage roles were resolved or publication rescanned native entrances");
            foreach(var p in roster) if(env.NativeDoorPresent(p,DwellKind.Break)!=(p.Kind==PlaceKind.Work)) throw new Exception("Work break or home/study eligibility changed");
            foreach(var b in CityManager.Instance.cityBuildingControllers) b.Destroyed=true;
            CityManager.Instance.cityBuildingControllers.Clear();
            foreach(var p in roster)
            {
                var b=new CityBuildingController {Entrance=p.Position}; b.buildingRegistration!.StreetNumber=CityManager.Instance.cityBuildingControllers.Count;
                CityManager.Instance.cityBuildingControllers.Add(b);
            }
            before=GamePlaces.EntranceReads;
            foreach(var p in roster) env.NativeDoorPresent(p,DwellKind.Break);
            if(env.FrontageRebindCount!=1 || GamePlaces.EntranceReads-before!=1000 || env.FrontageRebindLastMilliseconds<0
                || env.FrontageRebindTotalMilliseconds!=env.FrontageRebindLastMilliseconds
                || env.FrontageRebindMaximumMilliseconds!=env.FrontageRebindLastMilliseconds) throw new Exception("Rebind count/duration or one-image bound changed");
            before=GamePlaces.EntranceReads;
            foreach(var p in roster) env.NativeDoorPresent(p,DwellKind.Break);
            if(GamePlaces.EntranceReads!=before || env.FrontageRebindCount!=1) throw new Exception("Warm ownership reads republished the image");
            CityManager.Instance.cityBuildingControllers.Clear();
            Console.WriteLine($"Indexed frontage: {checks} three-axis negative/boundary/order checks; 1,000-controller mixed-role publish/rebind, work breaks, warm reuse and counters passed.");
        }

        static void CacheCosts(int count)
        {
            var venues=Enumerable.Range(0,count).Select(i=>new Place {Kind=PlaceKind.Work,Id="address:Test%20Street:"+i+":work",Position=new Position(i*10,0,0)}).ToArray();
            CityManager.Instance.cityBuildingControllers.Clear();
            foreach(var p in venues)
            {
                var controller=new CityBuildingController {Entrance=p.Position};
                controller.buildingRegistration!.StreetNumber=CityManager.Instance.cityBuildingControllers.Count;
                CityManager.Instance.cityBuildingControllers.Add(controller);
            }
            foreach(int model in new[]{0,1,2})
            {
                foreach(bool rebind in new[]{false,true})
                {
                    var times=new double[200]; long bytes=0; int reads=0;
                    for(int n=0;n<times.Length;n++)
                    {
                        var current=new NativeDoorHarness(); var baseline=new E2aDoorHarness(); var reviewed=new ReviewedDoorHarness();
                        // Both production constructors already own the place roster before the
                        // cache publication/rebind being measured. Keep fixture roster setup out.
                        foreach(var p in venues) current.places[p.Id]=p;
                        reviewed.SetRoster(venues);
                        if(rebind)
                        {
                            if(model==2) current.PublishFrontages(venues); else if(model==1) reviewed.Publish(); else baseline.Publish(venues);
                            foreach(var b in CityManager.Instance.cityBuildingControllers) b.Destroyed=true;
                            CityManager.Instance.cityBuildingControllers.Clear();
                            foreach(var p in venues)
                            {
                                var b=new CityBuildingController {Entrance=p.Position}; b.buildingRegistration!.StreetNumber=CityManager.Instance.cityBuildingControllers.Count;
                                CityManager.Instance.cityBuildingControllers.Add(b);
                            }
                        }
                        int beforeReads=GamePlaces.EntranceReads;
                        long before=GC.GetAllocatedBytesForCurrentThread(), start=System.Diagnostics.Stopwatch.GetTimestamp();
                        if(model==2)
                        {
                            if(rebind) foreach(var p in venues) current.NativeDoorPresent(p,DwellKind.Break);
                            else current.PublishFrontages(venues);
                        }
                        else if(model==1) { if(rebind) reviewed.ReadAll(); else reviewed.Publish(); }
                        else baseline.Publish(venues);
                        times[n]=System.Diagnostics.Stopwatch.GetElapsedTime(start).TotalMilliseconds;
                        bytes+=GC.GetAllocatedBytesForCurrentThread()-before; reads+=GamePlaces.EntranceReads-beforeReads;
                    }
                    Array.Sort(times);
                    foreach(double t in times.Where(t=>t>=30)) Console.WriteLine(FormattableString.Invariant($"FLAG >=30ms: cache model={model} rebind={rebind} count={count}; ms={t:F4}"));
                    Console.WriteLine(FormattableString.Invariant($"Venue cache model={model} rebind={rebind} count={count}: p50={times[99]:F4} p95={times[189]:F4} max={times[199]:F4} ms; bytes/batch={bytes/200.0:F1}; entrance reads/batch={reads/200.0:F1}."));
                }
            }
            // Repeated refusal cost: production verdicts, not a parallel matching algorithm.
            var absent=new Place {Kind=PlaceKind.Work,Id="missing",Position=new Position(-100,0,0)};
            var old=new E2aDoorHarness(); var next=new NativeDoorHarness(); next.Publish(new[]{absent});
            foreach(bool after in new[]{false,true})
            {
                int before=GamePlaces.EntranceReads; long bytes=GC.GetAllocatedBytesForCurrentThread();
                for(int n=0;n<200;n++) { bool value=after?next.NativeDoorPresent(absent,DwellKind.Break):old.Read(absent); if(value) throw new Exception("Missing frontage accepted"); }
                Console.WriteLine($"Cached refusal after={after} controllers={count}: 200 reads -> native entrance reads={GamePlaces.EntranceReads-before}, bytes={GC.GetAllocatedBytesForCurrentThread()-bytes}.");
            }
            CityManager.Instance.cityBuildingControllers.Clear();
        }
